\section*{Chapter 6: Adjectives}

    \setcounter{enumi}{1}
    \item Adjective: \textbf{long}, Kind: Adjective of Quality
    \item Adjective: \textbf{seven}, Kind: Adjective of Number
    \item Adjective: \textbf{enough}, Kind: Adjective of Quantity
\end{enumerate}
\subsection*{Activity B: Find the Adjectives}
\begin{enumerate}
    \setcounter{enumi}{1}
    \item \underline{many} (Number)
    \item \underline{wise} (Quality), \underline{old} (Quality)
    \item \underline{little} (Quantity)
    \item \underline{second} (Number)
\end{enumerate}
\subsection*{Activity C: Describe the Noun}

    \item My mother baked a \textbf{delicious} cake for my birthday.
    \item An elephant is a \textbf{large} animal.
    \item I have \textbf{five} pencils in my bag.
    \item Rohan is a \textbf{kind} and \textbf{helpful} boy.
\end{enumerate}
